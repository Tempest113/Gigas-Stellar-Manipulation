giga_systemcraft_enabled = {
    potential = {
        exists = from
        from = {
            has_country_flag = giga_menu_page_03
        }
        NOT = {
            has_global_flag = systemcraft_disabled
        }
    }

    allow = { always = yes }

    effect = {
        custom_tooltip = "giga_menu_disabled"
        set_global_flag = systemcraft_disabled
        set_global_flag = stellar_manip_o_systemcraft_disabled
        set_global_flag = stellar_manip_assembler_disabled
    }
}

celestial_printing_moon_x_disabled = {
    potential = {
        exists = from
        from = {
            has_country_flag = giga_menu_page_00
            NOT = { has_global_flag = giga_celestial_printing_moon_disabled }
        }
    }
    allow = {
        custom_tooltip = "must_not_have_started_game"
        hidden_trigger = { NOT = { has_global_flag = giga_game_started } }
    }
    effect = {
        custom_tooltip = "giga_celestial_printing_moon_enabled"
        hidden_effect = {
            set_global_flag = giga_celestial_printing_moon_disabled
            if = { #Planet Printer tech requires Moon Printer tech
                limit = {
                    NOT = {
                        has_global_flag = giga_celestial_printing_planet_disabled
                    }
                }
                set_global_flag = giga_celestial_printing_planet_disabled
            }
            # STELLAR MANIP: Systemcraft Assembler tech requires Planet Printer tech
            set_global_flag = stellar_manip_assembler_disabled
        }
    }
}
